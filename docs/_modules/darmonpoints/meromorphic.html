def evalpoly(poly, x, prec=None, check=True):
    if prec is None:
        prec = len(poly)
    # Initialize result
    if len(poly) == 0:
        return 0
    # Keep the low-degree terms t^0, ..., t^{prec-1} (matches the semantics
    # of the old call site `evalpoly(self._value[:prec], ...)`).
    poly = poly[:prec]
    K = x.parent()
    P0 = poly[0].parent()
    if P0 is not K:
        K = (P0(1) * K(1)).parent()
        x = K(x)
    # Coercing coefficients that already lie in K is not free (it goes through
    # the generic element constructor) and dominated the cost of evaluation.
    if P0 is not K or any(a.parent() is not K for a in poly):
        poly = [K(a) for a in poly]
    if check:
        assert x.valuation() >= 0
    # Evaluate value of polynomial using Horner's method:
    # result starts at 0 and we fold in poly[-1], poly[-2], ..., poly[0].
    # (Seeding result with a coefficient and then also including that same
    # coefficient in the loop double-counts it and adds a spurious
    # poly[-1] * x**len(poly) term -- that was the previous bug here.)
    result = K(0)
    for a in reversed(poly):
        result *= x
        result += a
    return result

class MeromorphicFunctionsElement(ModuleElement):
    def __init__(self, parent, data, parameter=None, check=True):
        if parameter is None:
            parameter = Matrix(parent.base_ring(), 2, 2, [1, 0, 0, 1])
        ModuleElement.__init__(self, parent)
        prec = parent._prec
        if check:
            K = parent.base_ring()
            Ps = parent._Ps
            self._value = None
            if data == 0:
                self._value = Ps(1).list()  # multiplicative!
            elif hasattr(data, "parent"):
                if isinstance(data.parent(), Divisors):
                    self._value = divisor_to_pseries(parameter, Ps, data, prec).list()
                    # assert len(data.support()) > 1
                elif data.parent() == parent:
                    self._value = data._value
                elif isinstance(data.parent(), MeromorphicFunctions):
                    self._value = vector(
                        list(data._value.apply_map(lambda x: x.lift_to_precision()))
                    )
            if self._value is None:
                val = Ps(data)
                val /= val[0]
                self._value = val.add_bigoh(prec).list()
        else:
            self._value = data
        self._moments = self._value
        self._value = [o.add_bigoh(prec) for o in self._value]
        while len(self._value) < prec:
            self._value.append(0)
        r = self._value[0]
        if r != 1:
            rinv = ~r
            self._value = [rinv * o for o in self._value]
        self._value = vector(self._value)
        self._parameter = Matrix(parent.base_ring(), 2, 2, parameter)

    def evaluate(self, D, prec=None):  # meromorphic functions
        K = self.parent().base_ring()
        if type(D) in (int, Integer):
            D = K(D)
        a, b, c, d = self._parameter.list()
        phi = lambda Q: a / c if Q == Infinity else (a * Q + b) / (c * Q + d)
        if isinstance(D.parent(), Divisors):
            return prod(evalpoly(self._value, phi(P), prec=prec) ** n for P, n in D)
        else:
            return evalpoly(self._value, phi(D), prec=prec)

    def eval_derivative(self, D, return_value=False):
        K = self.parent().base_ring()
        if type(D) in (int, Integer):
            D = K(D)
        if isinstance(D.parent(), Divisors):
            raise NotImplementedError
        K = self.parent().base_ring()
        a, b, c, d = self._parameter.list()
        phi = lambda Q: a / c if Q == Infinity else (a * Q + b) / (c * Q + d)
        phiD = phi(D)
        if return_value:
            val = evalpoly(self._value, phiD)

        valder = self._derivative_coeffs()
        chainrule = (a * d - b * c) / (c * D + d) ** 2
        ans = evalpoly(valder, phiD) * chainrule
        if return_value:
            return ans, val
        else:
            return ans

    def eval_dlog(self, D):
        K = self.parent().base_ring()
        if type(D) in (int, Integer):
            D = K(D)
        if isinstance(D.parent(), Divisors):
            raise NotImplementedError
        a, b, c, d = self._parameter.list()
        phi = lambda Q: a / c if Q == Infinity else (a * Q + b) / (c * Q + d)
        phiD = phi(D)
        # The coefficients of (log F)' do not depend on D: computing them (a power
        # series logarithm) dominated the cost of each call, so cache them.
        valder = self._dlog_coeffs()
        chainrule = (a * d - b * c) / (c * D + d) ** 2
        return chainrule * evalpoly(valder, phiD)



    def _cmp_(self, right):

    def valuation(self, p=None):
        K = self.parent().base_ring()
        a = (self._value[0] - 1).valuation(p)
        b = min([o.valuation(p) for o in self._value[1:]])
        return min(a, b)



    def _add_(self, right):  # multiplicative!
        if self._parameter != right._parameter:
            raise RuntimeError("Trying to add incompatible series")
        prec = self.parent()._prec
        ans = (self.power_series() * right.power_series()).list()[:prec]
        return self.__class__(self.parent(), ans, self._parameter, check=False)

    def _sub_(self, right):  # multiplicative!
        if self._parameter != right._parameter:
            raise RuntimeError("Trying to subtract incompatible series")
        prec = self.parent()._prec
        ans = (self.power_series() / right.power_series()).list()[:prec]
        return self.__class__(self.parent(), ans, self._parameter, check=False)

    def _neg_(self):  # multiplicative!
        prec = self.parent()._prec
        ans = (~self.power_series()).list()[:prec]
        return self.__class__(self.parent(), ans, self._parameter, check=False)

    def left_act_by_matrix(self, g, param=None, fast=True):  # meromorphic functions
        if param is None:
            param = self._parameter
        parent = self.parent()
        # Below we code the action which is compatible with the usual action
        # P |-> (aP+b)/(cP+D)
        if fast:
            zz_ps_vec = parent.get_action_data(g, self._parameter, param)
            ans = self._value * zz_ps_vec
            return self.__class__(parent, ans, param, check=False)
        else:
            set_immutable(g)
            assert g.parent() == MatrixSpace(parent.base_ring(), 2, 2), g.parent()
            K = parent.base_ring()
            prec = parent._prec
            assert param is not None
            tg = param.change_ring(K)
            # abcd = tau_0 g^-1 * tau_1^-1
            a, b, c, d = (self._parameter * (tg * g).adjugate()).list()
            zz = (
                (parent._Ps([b, a]) / parent._Ps([d, c]))
                .truncate(prec)
                .map_coefficients(lambda x: x.add_bigoh(prec + 1))
            )
            return self.__class__(parent, evalpoly(self._value, zz), param, check=False)

def divisor_to_pseries(parameter, Ps, data, prec):
    t = Ps.gen()
    a, b, c, d = parameter.list()
    num = Ps(1).truncate(prec)
    den = Ps(1).truncate(prec)
    for Q, n in data:
        K = Q.parent()
        if n > 0:
            num *= (1 - ((c * Q + d) / (a * Q + b)) * t) ** ZZ(n)
            num = num.add_bigoh(prec)
        else:
            den *= (1 - ((c * Q + d) / (a * Q + b)) * t) ** ZZ(-n)
            den = den.add_bigoh(prec)
    ans = Ps(num).add_bigoh(prec) * ~(Ps(den).add_bigoh(prec))
    return ans

    @cached_method
    def get_action_data(self, g, oldparam, param, prec=None):
        r"""
        Act on the power series by matrix g, given the old and new parameters.
        The action is given by the formula:
        (g * phi)(t) = phi(tau_0 g^-1 * tau_1^-1 t)
        which amounts to acting on the left by
        tau_1 g tau_0^-1
        """
        return self.compute_action_data(g, oldparam, param, prec)

    @muted
    def compute_action_data(self, g, oldparam, param, prec=None):
        r"""
        Uncached version of get_action_data. Looking up the cache of
        get_action_data needs hashing p-adic matrices, which is slow, so
        callers that do their own caching should use this one.
        """
        K = self.base_ring()
        if prec is None:
            prec = self._prec
        assert param is not None
        tg = param.change_ring(K)
        if not hasattr(g, "adjugate"):
            g = g.matrix().change_ring(K)

        # abcd = tau_0 g^-1 * tau_1^-1
        a, b, c, d = (oldparam * (tg * g).adjugate()).list()
        if not d.is_zero() and (c / d).valuation() >= 0:
            return self._action_matrix_moebius(a, b, c, d, prec)
        zz = (
            (self._Ps([b, a]) / self._Ps([d, c]))
            .truncate(prec)
            .map_coefficients(lambda x: x.add_bigoh(prec + 1))
        )  # zz = (at + b)/(ct + d)
        ans = [zz.parent()(1), zz]
        while len(ans) < prec:
            zz_ps = (
                (zz * ans[-1])
                .truncate(prec)
                .map_coefficients(lambda x: x.add_bigoh(prec + 1))
            )
            ans.append(zz_ps)
        m = Matrix(K, prec, prec, 0)
        for i, zz_ps in enumerate(ans):
            for j, aij in enumerate(zz_ps):
                m[i, j] = aij  # i, j entry contains j-th term of zz^i
        return m



    def _action_matrix_moebius(self, a, b, c, d, prec):
        r"""
        Same matrix as in get_action_data (row i contains the coefficients of
        zz^i, where zz = (at + b)/(ct + d)), computed using that
        zz^i = zz^(i-1) * (b1 + a1 t) / (1 + c1 t). This only needs O(prec)
        operations per row instead of a product of power series.

        Assumes that c/d has nonnegative valuation, so that the division by
        1 + c1 t does not lose precision.
        """
        K = self.base_ring()
        dinv = ~d
        a1, b1, c1 = a * dinv, b * dinv, c * dinv
        bound = prec + 1
        zero = K(0)
        m = Matrix(K, prec, prec, 0)
        m[0, 0] = 1
        w = [K(1)] + [zero] * (prec - 1)
        for i in range(1, prec):
            u = []
            prev_w = zero
            prev_u = zero
            for k in range(prec):
                wk = w[k]
                uk = b1 * wk + a1 * prev_w - c1 * prev_u
                if not uk._is_exact_zero():
                    uk = uk.add_bigoh(bound)
                prev_w = wk
                prev_u = uk
                u.append(uk)
            # Like the coefficients of a polynomial, drop the trailing zeros
            # (they are left as exact zeros in the matrix)
            n = prec
            while n > 0 and u[n - 1].is_zero():
                n -= 1
            if n == 0:
                break  # All the remaining powers vanish
            for j in range(n):
                m[i, j] = u[j]
            w = u
        return m

class MeromorphicIntegerArithmetic(SageObject):
    r"""
    Arithmetic of the power series of a :class:`MeromorphicFunctions` space
    (truncated to ``prec`` terms) with integers, as used to iterate Hecke
    operators quickly.

    The base field ``K`` must be `\QQ_p` or an unramified extension of it,
    and the series must have integral coefficients. An element of the ring
    of integers `O_K` is represented by its coordinates in the basis
    `1, g, \ldots, g^{d-1}` reduced modulo `p^N` (`N` = ``prec``), and a
    series by the list of the `d` integer polynomials (FLINT) of its
    coordinates. Since every operation is computed modulo `p^N`, the results
    agree with the `p`-adic computation done with absolute precision `N`
    (which is how :class:`MeromorphicFunctionsElement` works), but are much
    faster than products of `p`-adic numbers.

    The methods raise ``ValueError`` when the input is not integral or not
    known to precision `N`, or when a unit is needed and the element is not
    one, so that the caller can fall back to the `p`-adic computation.

    EXAMPLES::

        sage: from darmonpoints.meromorphic import MeromorphicFunctions, MeromorphicIntegerArithmetic
        sage: R.<x> = QQ[]
        sage: K.<g> = Qp(5, 10).extension(x^2 - 3)
        sage: M = MeromorphicFunctions(K, 5, 10)
        sage: I = MeromorphicIntegerArithmetic(M)
        sage: f = M([K(o) for o in [1, 5*g, 25, 0, 5^4*g + 5]], check=False)
        sage: h = M([K(o) for o in [1, 5, 5*g, 0, 0, 25]], check=False)
        sage: F, H = I.from_element(f), I.from_element(h)
        sage: (I.to_element(I.mul(F, H)) - (f + h)).valuation()
        10
        sage: (I.to_element(I.inverse(F)) - (-f)).valuation()
        10
        sage: A = matrix(K, 2, 2, [1, 2, 5, 3])
        sage: B = I.action_matrix(M.compute_action_data(A, f._parameter, f._parameter))
        sage: (I.to_element(I.normalize(I.act(F, B))) - f.left_act_by_matrix(A)).valuation()
        10
    """

    def __init__(self, M):
        from .theta_additive import _ok_coordinates

        K = M.base_ring()
        O = _ok_coordinates(K)
        if O.e != 1 or O._kind not in ("Qp", "unramified"):
            raise NotImplementedError(
                "Only implemented for Qp and its unramified extensions"
            )
        self._M = M
        self._O = O
        self.d = d = O.d
        self.n = n = ZZ(M._prec)
        self.p = p = ZZ(K.prime())
        self.N = p**n
        self._R = PolynomialRing(ZZ, names="t")
        # Structure constants: basis[r] * basis[s] = sum_l c[r, s, l] basis[l]
        self._structure = [
            (r, s, l, O.R[s][r, l])
            for r in range(d)
            for s in range(d)
            for l in range(d)
            if O.R[s][r, l] != 0
        ]
        assert O.coords(K(1)) == [1] + [0] * (d - 1)
        self.one = [self._R.one()] + [self._R.zero()] * (d - 1)

    def _red(self, F):
        N = self.N
        return [f % N for f in F]

    def _coords(self, x):
        if x.is_zero():
            if x.precision_absolute() < self.n:
                raise ValueError("Not enough precision")
            return [ZZ(0)] * self.d
        if x.precision_absolute() < self.n:
            raise ValueError("Not enough precision")
        try:
            c = self._O.coords(x)
        except (TypeError, ArithmeticError) as e:
            raise ValueError("Not integral") from e
        return [ZZ(o) % self.N for o in c]

    def from_element(self, f):
        r"""
        The integer representation of the series of ``f``.
        """
        n, d = self.n, self.d
        coords = [self._coords(o) for o in f._value[:n]]
        return [self._R([c[r] for c in coords]) for r in range(d)]

    def to_element(self, F, parameter=None):
        r"""
        The element of the space of meromorphic functions whose series is ``F``.
        """
        n, d, O = self.n, self.d, self._O
        lists = [f.padded_list(n) for f in F]
        vals = [O.element([lists[r][k] for r in range(d)], n) for k in range(n)]
        return self._M(vals, param=parameter, check=False)

    def mul(self, F, H):
        r"""
        The product of the series ``F`` and ``H`` (truncated).
        """
        n = self.n
        if self.d == 1:
            return [F[0]._mul_trunc_(H[0], n) % self.N]
        out = [self._R.zero()] * self.d
        for r, s, l, c in self._structure:
            out[l] += c * F[r]._mul_trunc_(H[s], n)
        return self._red(out)

    def scalar_inverse(self, c):
        r"""
        The coordinates of the inverse of the unit of `O_K` with coordinates ``c``.
        """
        x = self._O.element(c, self.n)
        if x.valuation() != 0:
            raise ValueError("Not a unit")
        return self._coords(~x)

    def inverse(self, F):
        r"""
        The inverse of the series ``F``, whose constant term must be a unit
        (Newton iteration).
        """
        R = self._R
        Y = self._scalar(self.scalar_inverse(self.constant_term(F)))
        k = 1
        while k < self.n:
            E = self.mul(F, Y)
            E = [-e for e in E]
            E[0] += 2
            Y = self.mul(Y, E)
            k *= 2
        return Y

    def normalize(self, F):
        r"""
        The multiple of ``F`` with constant term 1 (it must be a unit).
        """
        c = self.constant_term(F)
        if c == [1] + [0] * (self.d - 1):
            return F
        return self.mul(F, self._scalar(self.scalar_inverse(c)))

    def valuation(self, F):
        r"""
        The valuation of ``F - 1`` (at most `N`), as computed by
        :meth:`MeromorphicFunctionsElement.valuation`.
        """
        n, p = self.n, self.p
        G = list(F)
        G[0] = G[0] - 1
        ans = n
        for f in G:
            for c in f.list():
                if c != 0:
                    ans = min(ans, c.valuation(p))
        return ans

    def action_matrix(self, A):
        r"""
        The integer matrix of the action on the flattened coordinates of a
        series of the matrix ``A`` over `K` returned by
        :meth:`MeromorphicFunctions.get_action_data`.
        """
        n, d, N = self.n, self.d, self.N
        if A.nrows() != n or A.ncols() != n:
            raise ValueError("Wrong size")
        coords = [[self._coords(a) for a in row] for row in A.rows()]
        As = [
            matrix(ZZ, n, n, [[c[s] for c in row] for row in coords]) for s in range(d)
        ]
        if d == 1:
            return As[0]
        blocks = [[matrix(ZZ, n, n, 0) for l in range(d)] for r in range(d)]
        for r, s, l, c in self._structure:
            blocks[r][l] += c * As[s]
        return block_matrix(ZZ, blocks, subdivide=False) % N

    def act(self, F, B):
        r"""
        The action on the series ``F`` of the matrix ``B`` returned by
        :meth:`action_matrix` (not normalized).
        """
        n, d = self.n, self.d
        v = matrix(ZZ, 1, n * d, [c for f in F for c in f.padded_list(n)])
        w = ((v * B) % self.N).list()
        R = self._R
        return [R(w[r * n : (r + 1) * n]) for r in range(d)]
